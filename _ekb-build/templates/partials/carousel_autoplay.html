{#- Автопрокрутка ленты. Подключается внутрь скрипта карусели (reviews.html, works.html) и берёт оттуда track, n и current(). Пауза между сменами — переменная delay из include (отзывы 5 с, фото доставок 3 с). Лента сама переходит к следующему слайду, после последнего — к первому. Стоит, только пока ленту держат пальцем или мышью, и ещё 5 секунд после свайпа или стрелки, чтобы не выдёргивать слайд, который человек сам только что открыл. Вне экрана и в фоновой вкладке не крутится. Смена — плавным сдвигом: текущий слайд уезжает влево, следующий въезжает справа за 0,7 с, потом лента без анимации встаёт на него. Анимация своя (Web Animations), а не scrollTo smooth: тот браузер крутит рывком за долю секунды, на телефонах с «уменьшить движение» (оно часто включено вместе с энергосбережением) сдвигает вовсе без анимации, а с последнего на первый проматывал ленту назад через все слайды. Порядок важен: пока слайды сдвинуты, браузер // считает точки прилипания по сдвинутым местам и отщёлкивает ленту. m.anims.forEach(function(a){ a.cancel(); }); track.scrollTo({ left: m.j * track.clientWidth, behavior: 'instant' }); track.style.scrollSnapType = ''; } function slideTo(j){ var i = current(), from = track.children[i], to = track.children[j]; if(!to || i === j) return; if(!to.animate){ track.scrollTo({ left: j * track.clientWidth, behavior: 'smooth' }); return; } moving = { j: j, anims: [] }; var m = moving; ready(to).then(function(){ if(moving !== m) return; // Следующий слайд ставим сразу справа от видимого: для соседнего // сдвиг 0, для первого после последнего — на всю длину ленты. var shift = (i + 1 - j) * 100; track.style.scrollSnapType = 'none'; var opt = { duration: SLIDE, easing: EASE, fill: 'forwards' }; m.anims = [ from.animate([{ transform: 'translateX(0)' }, { transform: 'translateX(-100%)' }], opt), to.animate([{ transform: 'translateX(' + shift + '%)' }, { transform: 'translateX(' + (shift - 100) + '%)' }], opt) ]; m.anims[1].onfinish = finish; }); } setInterval(function(){ // Пока ленту не видно или её держат, отсчёт начинается заново: // первый слайд после появления на экране стоит полную паузу. if(!seen || held || focus || document.hidden){ wait(DELAY); return; } if(moving || Date.now() < due) return; var i = current(); slideTo(i >= n - 1 ? 0 : i + 1); wait(DELAY + SLIDE); }, 250); // Видна ли лента хотя бы на треть.
